Code : ip.py
ip = input("Adresse ip ? ").split(".")
if int(ip[3]) == 0:
    print("Adresse réseau")
elif int(ip[3]) == 255:
    print("Adresse broadcast")
else:
    print("Adresse machine")
Code : liste.py
n = int(input("n ?"))
l = []
for i in range (n):
    l.append(int(input(f"l[{i}]= ")))
print(f"Liste créer : {l}")
lm = sum(l)
lm2 = lm / n
print(f"La moyenne est de {round (lm2, 2)}")

n = int(input("n ? "))
for i in range (1):
    print("*"*n)
    for i in range(n-2):
        print("*","*", sep=" "*int(n-2))
    print("*"*n)
